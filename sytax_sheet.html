02

Registers

// Think of registers as ultra-fast variables inside the CPU itself.

register sizes — same physical register, different widths
; ┌──────────────── EAX  (32-bit) ─────────────────┐
; │                ┌──────── AX (16-bit) ──────────┐│
; │                │  AH (8-bit) │  AL (8-bit)     ││
; └────────────────┴─────────────┴─────────────────┘│
; Same pattern: EBX/BX/BH/BL  ECX/CX/CH/CL  EDX/DX/DH/DL
EAX accumulator

Default for arithmetic. MUL/DIV result stored here automatically. Used for I/O & interrupt calls.

mov eax, 5
add eax, 3    ; eax = 8
mul ebx      ; eax = eax × ebx
EBX base register

Memory base pointer. Temp storage when EAX is busy. Used for memory access.

mov ebx, arr    ; point to array
mov eax, [ebx]  ; read value at *ebx
ECX counter

LOOP auto-decrements ECX. CL holds shift amount for SHL/SHR.

mov ecx, 5     ; loop 5 times
shl eax, cl   ; shift by CL bits
EDX data register

Holds HIGH 32 bits of MUL result. EDX:EAX together = 64-bit result.

; After MUL instruction:
; EDX = high 32 bits
; EAX = low  32 bits
other registers
RegisterNamePurpose
ESI / SISource IndexHigh-speed memory transfer (source pointer)
EDI / DIDestination IndexHigh-speed memory transfer (destination pointer)
ESP / SPStack PointerPoints to top of stack — do NOT use for arithmetic
EBP / BPFrame PointerReferences function params and local vars on stack
EIPInstruction PointerAddress of the NEXT instruction to execute
EFLAGSFlags RegisterCF, ZF, SF, OF, PF — auto-set after arithmetic

endianness — x86 uses LITTLE endian
; Storing value: 0x0A0B0C0D

; Big Endian (IBM, Motorola, Sun):
addr+0:  0A  ← MSB first
addr+1:  0B

; Little Endian (Intel x86, DEC):
addr+0:  0D  ← LSB first

; myDouble DWORD 12345678h stored in memory (little endian):
; addr+0: 78  addr+1: 56  addr+2: 34  addr+3: 12

myDouble DWORD 12345678h
mov ax, WORD PTR myDouble   ; AX = 5678h  ← reads LOW order bytes first!
; NOT 1234h — because lowest address holds 78, 56
📌 x86 = little endian. The SMALLEST (lowest) byte goes at the SMALLEST memory address. That's why WORD PTR gives you 5678h not 1234h.
04

MOV / MOVZX / MOVSX

// MOV copies data. MOVZX and MOVSX extend a smaller value into a larger register.

MOV — copy source → destination
mov eax, 42           ; register ← immediate
mov eax, ebx          ; register ← register
mov eax, [var]        ; register ← memory
mov [var], eax        ; memory   ← register

; ❌ mov [a], [b]   — two memory operands NOT allowed
; ❌ mov eax, al    — sizes must match
; ❌ mov al, 300    — value too big for 8-bit register
MOVZX — zero extend (for unsigned)
; AL = 10001111b
movzx eax, al
; EAX = 00000000 00000000 00000000 10001111
;       ←── fills with 0s ───────────────

movzx ax,  al    ; byte  → word
movzx eax, al    ; byte  → dword
movzx eax, ax    ; word  → dword
MOVSX — sign extend (for signed)
; AL = 10001111b  (MSB=1 → negative)
movsx eax, al
; EAX = 11111111 11111111 11111111 10001111
;       ←── fills with sign bit (1) ─────

movsx ax,  al    ; byte  → word
movsx eax, al    ; byte  → dword
movsx eax, bx    ; EAX = FFFFA69Bh (if BX=A69Bh)
workaround before MOVZX existed (manual zero-extend)
; Moving 16-bit "count" into 32-bit ECX:
count WORD 1
mov ecx, 0     ; clear top 16 bits
mov cx, count  ; copy into low 16 bits
📌 MOVZX = fill with 0s (unsigned numbers). MOVSX = fill with sign bit (signed numbers). When in doubt: if the number can be negative → use MOVSX.

Arithmetic Instructions

// ADD, SUB, MUL, DIV, INC, DEC — and shift instructions.

all arithmetic instructions
InstructionSyntaxEffect
ADDadd eax, srceax = eax + src
SUBsub eax, srceax = eax − src
MULmul srcEDX:EAX = EAX × src (unsigned, no dest needed)
IMULimul eax, srceax = eax × src (signed)
DIVdiv srcEAX = EDX:EAX ÷ src  |  EDX = remainder
INCinc eaxeax++  (does NOT change CF)
DECdec eaxeax--  (does NOT change CF)
NEGneg eaxeax = −eax  (two's complement)
shift / rotate instructions (ECX / CL)
shl dst, clshift left = multiply by 2ⁿ
shr dst, clshift right = divide by 2ⁿ (unsigned)
sar dst, clarithmetic shift right (preserves sign)
rol dst, clrotate left
ror dst, clrotate right
example — int x = (y + 4) × 3
mov eax, y    ; eax = y
add eax, 4    ; eax = y + 4
mov ebx, 3    ; ebx = 3
mul ebx       ; eax = (y+4) × 3
mov x, eax    ; store result
⚠ MUL always uses EAX automatically. You only specify the second operand. Result goes into EDX:EAX (EDX = overflow bits).

Boolean / Bitwise Instructions

// AND, OR, XOR, NOT, TEST — for bit manipulation.

AND — both bits must be 1
and al, 11110000b  ; clear low 4 bits
and ax, 00FFh      ; clear high byte of AX
; Clears OF and CF
Mask with 0 to CLEAR bits.
OR — either bit can be 1
or al, 00001111b   ; set low 4 bits to 1
or al, 80h         ; force bit 7 = 1
; Clears OF and CF
Mask with 1 to SET bits.
XOR — bits differ = 1
xor eax, eax       ; fastest way to clear EAX = 0
xor al, key        ; encrypt byte
xor al, key        ; decrypt (XOR same key twice = original)
XOR a value with itself = 0. XOR twice = original.
NOT and TEST
not al             ; flip ALL bits (no flags affected)

test al, 00001001b ; check bits 0 and 3
; like AND but does NOT change AL
; only sets ZF, SF, PF
practical tricks from lecture
; lowercase → uppercase (clear bit 5)
and al, 11011111b   ; 'a'(61h) → 'A'(41h)

; uppercase → lowercase (set bit 5)
or  al, 00100000b   ; 'A'(41h) → 'a'(61h)

; fastest register clear
xor eax, eax        ; eax = 0 (faster than mov eax,0)

; test if bit 3 is set, jump if yes
test al, 00001000b
jnz  bit3_set       ; ZF=0 means bit 3 = 1

; set intersection (AND) and union (OR)
and eax, SetY       ; SetX ∩ SetY
or  eax, SetY       ; SetX ∪ SetY
not eax             ; complement of SetX

XCHG / NEG

// XCHG swaps two values. NEG negates (flips sign).

XCHG — exchange / swap
xchg ax, bx       ; swap 16-bit regs
xchg ah, al       ; swap 8-bit regs
xchg eax, ebx     ; swap 32-bit regs
xchg var1, bx     ; swap mem ↔ register

; XCHG reg, reg
; XCHG reg, mem
; XCHG mem, reg
; ❌ does NOT accept immediate values (unlike MOV)
NEG — negate (two's complement)
mov al, -1
neg al            ; AL = 1  (+1)

mov al, 127
neg al            ; AL = -127

neg eax           ; works on 32-bit too
neg [var]         ; works on memory

; NEG reg
; NEG mem
; Applying NEG to non-zero always sets CF!
⚠ NEG on -128 (10000000b) causes OF=1 because +128 doesn't fit in a signed byte (max is +127).

CPU Flags (EFLAGS register)

// Automatically set/cleared after arithmetic. You check them with conditional jumps.

status flags
FlagNameSet when...
CFCarry FlagUnsigned overflow — result too big for destination (carry out of MSB)
ZFZero FlagResult = 0
SFSign FlagResult is negative (MSB = 1)
OFOverflow FlagSigned overflow/underflow (e.g. 127 + 1 = -128)
PFParity FlagLow byte of result has EVEN number of 1-bits
AFAuxiliary CarryCarry from bit 3 → bit 4 (used in BCD arithmetic)
flag examples
mov al, 0FFh
add al, 1       ; AL=0,  CF=1, ZF=1

mov eax, 4
sub eax, 5      ; EAX=-1, SF=1

mov al, 127
add al, 1       ; AL=-128, OF=1 (signed overflow)

mov ecx, 1
sub ecx, 1      ; ECX=0, ZF=1
set / clear flags manually
stc             ; SET   Carry Flag
clc             ; CLEAR Carry Flag

or  al, 80h    ; SET   Sign Flag  (set bit 7)
and al, 7Fh    ; CLEAR Sign Flag  (clear bit 7)

test al, 0     ; SET   Zero Flag
or   al, 1     ; CLEAR Zero Flag

mov  al, 7Fh
inc  al         ; SET   Overflow Flag (127+1)
or   eax, 0    ; CLEAR Overflow Flag
📌 INC and DEC do NOT affect the Carry Flag. NEG on a non-zero value always SETS the Carry Flag.

CMP & Conditional Jumps

// CMP sets flags. Then a Jxx instruction checks those flags and jumps or not.

CMP — implied subtraction (sets flags, changes nothing)
cmp destination, source   ; internally does: destination - source
cmp eax, 10              ; sets CF, ZF, SF, OF based on eax-10
cmp ax, bx               ; registers are NOT changed by CMP
jump on flag value
JZ / JE ZF=1 (zero / equal)
JNZ / JNEZF=0 (not zero / not equal)
JC CF=1 (carry set)
JNC CF=0 (no carry)
JS SF=1 (negative result)
JNS SF=0 (positive result)
JO OF=1 (overflow)
JP PF=1 (even parity)
signed comparisons (after CMP)
JG / JNLEgreater (dst > src)
JGE / JNLgreater or equal (≥)
JL / JNGEless (dst < src)
JLE / JNGless or equal (≤)

unsigned comparisons (after CMP)
JA / JNBEabove (dst > src, unsigned)
JAE / JNBabove or equal
JB / JNAEbelow (dst < src, unsigned)
JBE / JNAbelow or equal
equality jumps
JE equal (leftOp = rightOp)
JNE not equal (leftOp ≠ rightOp)
JCXZ jump if CX = 0
JECXZ jump if ECX = 0
JMP lbl unconditional (always jumps)
CMP results — unsigned
CMP resultZFCF
dst < src01
dst > src00
dst = src10
if-else pattern in assembly
; if (eax == 10) { doA } else { doB }
    cmp  eax, 10
    jne  else_block    ; jump if NOT equal
    ; === IF block ===
    jmp  done
else_block:
    ; === ELSE block ===
done:

LOOP Instruction

// LOOP auto-decrements ECX and jumps back if ECX ≠ 0. Built-in for-loop counter.

LOOP syntax
mov ecx, 5        ; loop 5 times
myLoop:
    ; body here runs 5 times
    loop myLoop   ; ECX-- ; if ECX≠0 → jump back

; What LOOP does each iteration:
; step 1: ECX = ECX - 1
; step 2: if ECX ≠ 0 → jump to label
; step 3: if ECX = 0 → fall through (exit loop)
LOOP variants
LOOP lbljump if ECX ≠ 0
LOOPE lbljump if ECX ≠ 0 AND ZF = 1
LOOPNE lbljump if ECX ≠ 0 AND ZF = 0

manual loop with JMP
mov ecx, 0
top:
    cmp ecx, 5
    jge done      ; exit when ecx >= 5
    ; body
    inc ecx
    jmp top
done:
⚠ If ECX = 0 before LOOP: it wraps to 4,294,967,295 and runs ~4 billion times! Always initialize ECX first.
saving ECX inside a loop body
mov ecx, 100
top:
    mov count, ecx    ; save ECX before changing it
    mov ecx, 20       ; inner use
    ; ...do stuff...
    mov ecx, count    ; restore ECX
    loop top

Addressing Modes

// 5 ways to specify where your data lives.

1. immediate — value is inside the instruction
mov al, 123      ; 123 IS the data, no memory needed
add eax, 5       ; 5 is immediate
; fastest mode, no memory access required
2. direct — variable name (constant address)
mov al, var1      ; CPU goes directly to var1's address
mov eax, [10400h] ; or raw hex address
; machine code encodes the full address
3. indirect — register holds the address (pointer)
mov esi, OFFSET byteVal
mov al,  [esi]     ; read from address in ESI
mov [esi], bl      ; write to address in ESI
inc BYTE PTR [esi] ; must specify size with PTR!
4. relative — PC + offset (used for jumps)
jmp 4             ; PC = PC + 4
jmp myLabel       ; assembler calculates offset
; used for: switch, virtual functions,
; function pointers, dynamic libraries
5. displacement / index — base + constant offset
arr BYTE 10h,20h,30h

mov al, arr        ; AL = 10h (element 0)
mov al, [arr+1]    ; AL = 20h (element 1)
mov al, [arr+2]    ; AL = 30h (element 2)
; arr+1 is called the "effective address"
GP fault warning
; ❌ If ESI is uninitialized:
mov ax, [esi]      ; GENERAL PROTECTION FAULT!

; ✅ Always initialize before using as pointer:
mov esi, OFFSET byteVal
mov al, [esi]      ; safe

arr WORD 1,2,3,4,5

intArray WORD 32 DUP(0)
SIZEOF intArray ; = 64  (32 × 2 bytes)
ALIGN / LABEL

ALIGN 2         ; next var on even address

Arrays

// Stepping through arrays using indirect and displacement addressing.

BYTE array — step by 1
arrayB BYTE 10h, 20h, 30h

mov esi, OFFSET arrayB
mov al, [esi]     ; AL = 10h
inc esi
mov al, [esi]     ; AL = 20h
inc esi
mov al, [esi]     ; AL = 30h
WORD array — step by 2
arrayW WORD 1000h, 2000h, 3000h

mov esi, OFFSET arrayW
mov ax, [esi]     ; AX = 1000h
add esi, 2
mov ax, [esi]     ; AX = 2000h
add esi, 2
mov ax, [esi]     ; AX = 3000h
DWORD array — step by 4
arrayD DWORD 10000h, 20000h, 30000h

mov esi, OFFSET arrayD
mov eax, [esi]    ; EAX = 10000h
add esi, 4
mov eax, [esi]    ; EAX = 20000h
displacement (index) access
arrayB BYTE 10h,20h,30h,40h,50h

mov al, arrayB      ; AL = 10h (index 0)
mov al, [arrayB+1]  ; AL = 20h (index 1)
mov al, [arrayB+2]  ; AL = 30h (index 2)
; arrayB+1 = "effective address"
sum array with loop (lecture example)
intarray DWORD 10000h,20000h,30000h,40000h

mov edi, OFFSET intarray      ; 1. pointer to array
mov ecx, LENGTHOF intarray    ; 2. ECX = 4 (element count)
mov eax, 0                    ; 3. sum = 0
L1:
    add eax, [edi]             ; 4. sum += *edi
    add edi, TYPE intarray     ; 5. edi += 4 (next element)
    loop L1                   ; 6. ECX-- ; repeat
📌 Use TYPE intarray to advance the pointer — it automatically gives the right step size (1 for BYTE, 2 for WORD, 4 for DWORD).

Stack / PUSH / POP

// Stack grows DOWNWARD in memory. LIFO — last pushed = first popped.

PUSH and POP
push eax       ; ESP -= 4; [ESP] = EAX
push 42        ; push an immediate value
pop  eax       ; EAX = [ESP]; ESP += 4
pop  ebx       ; pop into a different register
CALL and RET
call myFunc    ; pushes return addr, jumps to myFunc
ret            ; pops return addr, jumps back

; Standard function frame:
push ebp
mov  ebp, esp  ; set up stack frame
; ... function body ...
pop  ebp
ret
visual — stack memory (grows downward)
; Higher memory address ─────────────── bottom of stack
;   [ old data                                         ]
;   [ value pushed first                               ]
;   [ value pushed second                              ]
;   [ value pushed last   ] ← ESP points here (TOP)
; Lower memory address  ──────────────── top of stack

Program Skeleton

// Minimum structure every MASM/NASM program needs.

complete program structure
section .data
    a      DWORD  5
    b      DWORD  3
    result DWORD  0

section .bss
    buf    RESD   1          ; reserve 1 dword slot

section .text
    global _start

_start:
    mov eax, [a]
    add eax, [b]
    mov [result], eax

    mov eax, 1              ; syscall 1 = exit
    xor ebx, ebx            ; exit code 0
    int 0x80                ; call the kernel
print a string to screen (Linux syscall)
section .data
    msg db "Hello!", 0xA    ; 0xA = newline character
    len equ $ - msg          ; auto-calculate string length

_start:
    mov eax, 4              ; syscall 4 = sys_write
    mov ebx, 1              ; 1 = stdout (screen)
    mov ecx, msg            ; pointer to the string
    mov edx, len            ; number of bytes to print
    int 0x80                ; ← this actually prints it!
📌 3 sections every program needs: .data (vars with values) → .bss (empty slots) → .text (your code). Entry point = _start.

// All activities from lectures 1–5 with complete solutions.

activity 1 — flag values after each instruction
mov al, 00001111b
test al, 00000010b    ; CF=0  ZF=1  SF=0   (result=0, no bits overlap)

activity 2 — clear high 8 bits of AX, keep low 8 bits
and ax, 00FFh    ; clears AH (high byte), preserves AL (low byte)
; 00FFh = 00000000 11111111 — zeros in top, ones in bottom
activity 3 — sum DWORD array with loop
intarray DWORD 10000h,20000h,30000h,40000h

mov edi, OFFSET intarray    ; EDI = address of array
mov ecx, LENGTHOF intarray  ; ECX = 4
mov eax, 0                  ; sum = 0
L1:
    add eax, [edi]           ; sum += current element
    add edi, TYPE intarray   ; edi += 4 (next dword)
    loop L1                 ; ECX-- ; repeat until 0
; EAX = A0000h (result)
activity 4 — copy string with loop
source BYTE "This is the source string",0
target BYTE SIZEOF source DUP(0)

mov esi, 0              ; index = 0
mov ecx, SIZEOF source  ; loop counter
L1:
    mov al, source[esi]  ; get char from source
    mov target[esi], al  ; store in target
    inc esi
    loop L1
; Note: MOV can't have two memory operands, so AL is the middleman
activity 5 — convert array to uppercase
array BYTE "This Sentence is in Mixed Case", 0

mov ecx, LENGTHOF array
mov esi, OFFSET array
L1:
    and BYTE PTR [esi], 11011111b  ; clear bit 5 → uppercase
    inc esi
    loop L1
; Bit 5 = 0 → uppercase.  Bit 5 = 1 → lowercase
activity 6 — nested loop: what is final EAX?
mov eax, 0
mov ecx, 10
L1:
    mov eax, 3      ; ← resets EAX to 3 EVERY outer iteration
    mov ecx, 5      ; ← OVERWRITES ECX! outer loop loses its count
L2:
    add eax, 5
    loop L2        ; runs 5 times: EAX = 3 + (5×5) = 28
    loop L1        ; ECX is now 0 after L2, so L1 runs only ONCE

; Answer: EAX = 28
; Why: inner loop overwrites ECX=5, L2 runs 5×, then ECX=0 → L1 exits immediately
